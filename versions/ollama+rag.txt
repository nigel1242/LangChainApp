import streamlit as st
import ollama
import os
import sqlite3
import gc
import time
from pypdf import PdfReader

from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import FAISS
from langchain_core.embeddings import Embeddings
from langchain_core.documents import Document

# ------------------ CONFIG ------------------
st.set_page_config(
    page_title="AI Playground with RAG",
    page_icon="💬",
    layout="wide",
)

CHAT_DB_FILE = "chat_playground.db"
RAG_INDEX_DIR = "rag_indices"
os.makedirs(RAG_INDEX_DIR, exist_ok=True)

# ------------------ DATABASE ------------------
def init_db():
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS chats (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            model_name TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS messages (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            chat_id INTEGER NOT NULL,
            role TEXT NOT NULL,
            content TEXT NOT NULL,
            used_rag INTEGER DEFAULT 0,
            FOREIGN KEY (chat_id) REFERENCES chats (id)
        )
    """)
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS rag_docs (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            chat_id INTEGER NOT NULL,
            file_name TEXT,
            content TEXT,
            FOREIGN KEY (chat_id) REFERENCES chats (id)
        )
    """)
    conn.commit()
    conn.close()

def create_new_chat(model_name: str, first_message: str = ""):
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute("INSERT INTO chats (model_name) VALUES (?)", (model_name,))
    chat_id = cursor.lastrowid
    if first_message:
        cursor.execute(
            "INSERT INTO messages (chat_id, role, content) VALUES (?, ?, ?)",
            (chat_id, "user", first_message),
        )
    conn.commit()
    conn.close()
    return chat_id

def add_message(chat_id, role, content, used_rag=0):
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute(
        "INSERT INTO messages (chat_id, role, content, used_rag) VALUES (?, ?, ?, ?)",
        (chat_id, role, content, used_rag),
    )
    conn.commit()
    conn.close()

def load_all_chats():
    if not os.path.exists(CHAT_DB_FILE):
        return []
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute("SELECT id, model_name, created_at FROM chats ORDER BY created_at DESC")
    rows = cursor.fetchall()
    conn.close()
    return rows

def load_chat(chat_id):
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute("SELECT role, content, used_rag FROM messages WHERE chat_id = ? ORDER BY id ASC", (chat_id,))
    rows = cursor.fetchall()
    cursor.execute("SELECT model_name FROM chats WHERE id = ?", (chat_id,))
    model_name = cursor.fetchone()[0]
    conn.close()
    return [{"role": r, "content": c, "used_rag": ur} for r, c, ur in rows], model_name

def add_rag_doc(chat_id, file_name, content):
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute(
        "INSERT INTO rag_docs (chat_id, file_name, content) VALUES (?, ?, ?)",
        (chat_id, file_name, content),
    )
    conn.commit()
    conn.close()

def get_chat_documents(chat_id):
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute("SELECT file_name FROM rag_docs WHERE chat_id = ?", (chat_id,))
    rows = cursor.fetchall()
    conn.close()
    return [row[0] for row in rows]

# ------------------ RAG / Ollama ------------------
class OllamaEmbeddings(Embeddings):
    def __init__(self, model_name):
        self.model_name = model_name

    def embed_documents(self, texts):
        return [ollama.embeddings(model=self.model_name, prompt=t)['embedding'] for t in texts]

    def embed_query(self, query):
        return ollama.embeddings(model=self.model_name, prompt=query)['embedding']

def rebuild_rag_index(chat_id, model_name):
    conn = sqlite3.connect(CHAT_DB_FILE)
    cursor = conn.cursor()
    cursor.execute("SELECT content, file_name FROM rag_docs WHERE chat_id = ?", (chat_id,))
    rows = cursor.fetchall()
    conn.close()
    if not rows:
        return

    all_docs = []
    splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    for content, file_name in rows:
        chunks = splitter.split_text(content)
        for chunk in chunks:
            all_docs.append(Document(page_content=chunk, metadata={"source": file_name}))

    if not all_docs:
        return

    embeddings = OllamaEmbeddings(model_name)
    faiss_index = FAISS.from_documents(all_docs, embeddings)
    faiss_index.save_local(os.path.join(RAG_INDEX_DIR, f"chat_{chat_id}.faiss"))

def get_relevant_rag(chat_id, query, model_name, top_k=3):
    faiss_index_path = os.path.join(RAG_INDEX_DIR, f"chat_{chat_id}.faiss")
    if not os.path.exists(faiss_index_path):
        return "", False
    embeddings = OllamaEmbeddings(model_name)
    faiss_index = FAISS.load_local(faiss_index_path, embeddings, allow_dangerous_deserialization=True)
    docs_with_scores = faiss_index.similarity_search_with_score(query, k=top_k)
    if not docs_with_scores:
        return "", False
    relevant_docs = [doc for doc, score in docs_with_scores]  # take all top-k
    context = "\n\n".join(f"Source: {d.metadata.get('source','unknown')}\nContent: {d.page_content}" for d in relevant_docs)
    return context, True

# ------------------ PDF Helper ------------------
def extract_text_from_pdf(file_bytes):
    reader = PdfReader(file_bytes)
    text_content = ""
    for page in reader.pages:
        page_text = page.extract_text()
        if page_text:
            text_content += page_text + "\n"
    return text_content

# ------------------ MAIN APP ------------------
def main():
    st.title("💬 AI Playground with RAG (Ollama + OpenAI)")
    init_db()

    # Session state
    if "messages" not in st.session_state:
        st.session_state.messages = []
    if "current_chat_id" not in st.session_state:
        st.session_state.current_chat_id = None
    if "selected_model" not in st.session_state:
        st.session_state.selected_model = None
    if "uploaded_files" not in st.session_state:
        st.session_state.uploaded_files = []

    # --- Sidebar ---
    with st.sidebar:
        st.header("💾 Chat Sessions")
        if st.button("🆕 New Chat"):
            st.session_state.messages = []
            st.session_state.current_chat_id = None
            st.session_state.selected_model = None
            st.session_state.uploaded_files = []

        for chat_id, model_name, _ in load_all_chats():
            if st.button(f"{model_name} Chat", key=f"chat_{chat_id}"):
                msgs, model = load_chat(chat_id)
                st.session_state.messages = msgs
                st.session_state.current_chat_id = chat_id
                st.session_state.selected_model = model

        st.markdown("---")
        st.session_state.rag_mode = st.checkbox("💡 Enable RAG (Ollama only)", value=True)

    # Available Ollama models
    try:
        ollama_models = tuple(m['model'] for m in ollama.list().get("models", []))
    except Exception:
        ollama_models = ()
    openai_models = ("gpt-3.5-turbo", "gpt-4")
    available_models = ollama_models + openai_models

    if not available_models:
        st.warning("⚠️ No models available.")
        st.stop()

    selected_model = st.selectbox("🧠 Choose model", available_models)
    st.session_state.selected_model = selected_model

    message_container = st.container()

    # --- RAG File Upload ---
    with st.expander("📤 Upload Documents"):
        uploaded = st.file_uploader(
            "Upload TXT or PDF", type=["txt", "pdf"], accept_multiple_files=True
        )
        if uploaded:
            st.session_state.uploaded_files = uploaded

        if st.session_state.uploaded_files:
            if st.button("📂 Process Files"):
                chat_id = st.session_state.current_chat_id
                if chat_id is None:
                    chat_id = create_new_chat(st.session_state.selected_model)
                    st.session_state.current_chat_id = chat_id
                for f in st.session_state.uploaded_files:
                    content = ""
                    if f.type == "application/pdf":
                        content = extract_text_from_pdf(f)
                    else:
                        content = f.read().decode("utf-8")
                    add_rag_doc(chat_id, f.name, content)
                    add_message(chat_id, "system", f"📄 File uploaded: {f.name}", used_rag=1)
                rebuild_rag_index(chat_id, st.session_state.selected_model)
                st.success("✅ Files indexed!")
                st.session_state.uploaded_files = []

    # --- Display chat history ---
    for msg in st.session_state.messages:
        avatar = "🤖" if msg["role"]=="assistant" else "😎"
        prefix = "📄 " if msg.get("used_rag") else ""
        with message_container.chat_message(msg["role"], avatar=avatar):
            st.markdown(prefix + msg["content"])

    # --- Chat input ---
    if prompt := st.chat_input("Type your message here..."):
        chat_id = st.session_state.current_chat_id or create_new_chat(st.session_state.selected_model, prompt)
        st.session_state.current_chat_id = chat_id
        add_message(chat_id, "user", prompt)
        st.session_state.messages.append({"role":"user","content":prompt,"used_rag":0})

        with message_container.chat_message("user", avatar="😎"):
            st.markdown(prompt)

        # --- Only Ollama uses RAG ---
        use_rag = False
        rag_content = ""
        if st.session_state.rag_mode and st.session_state.selected_model in ollama_models:
            rag_content, has_docs = get_relevant_rag(chat_id, prompt, st.session_state.selected_model)
            use_rag = has_docs

        enhanced_prompt = prompt
        if use_rag:
            enhanced_prompt = f"""You have access to relevant background documents.
Answer naturally using the documents if possible.

Background documents:
{rag_content}

Question: {prompt}"""

        # --- Generate response ---
        avatar = "🤖"
        with message_container.chat_message("assistant", avatar=avatar):
            stream_placeholder = st.empty()
            streamed_text = ""
            try:
                for chunk in ollama.chat(model=st.session_state.selected_model,
                                         messages=[{"role":"user","content":enhanced_prompt}],
                                         stream=True):
                    delta = chunk.get("message", {}).get("content", "") or chunk.get("delta", "")
                    if delta:
                        streamed_text += delta
                        display_text = ("📄 " if use_rag else "") + streamed_text
                        stream_placeholder.markdown(display_text)

                add_message(chat_id, "assistant", streamed_text, used_rag=int(use_rag))
                st.session_state.messages.append({"role":"assistant","content":streamed_text,"used_rag":int(use_rag)})
            except Exception as e:
                st.error(f"Error: {e}")
                add_message(chat_id, "assistant", f"[ERROR] {e}", used_rag=0)

if __name__ == "__main__":
    main()
